const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");

const WIDTH = 400;
const HEIGHT = 450;

// Game state
let state = "start"; // start, playing, gameover, victory

// Player
let player = { x: 180, y: 380, w: 40, h: 30, speed: 0 };

// Laser
let laser = { x: 200, y: -20, w: 4, h: 10, speed: -10 };

// Invaders
let invaders = [];
let colors = ["red", "orange", "yellow"];
let invaderDX = 3;
let invaderDY = 20;

// Initialize enemies
function initInvaders() {
  invaders = [];
  for (let row = 0; row < 3; row++) {
    for (let col = 0; col < 5; col++) {
      invaders.push({
        x: 80 + col * 50,
        y: 80 + row * 40,
        r: 12,
        color: colors[row]
      });
    }
  }
}

initInvaders();

// Controls
document.addEventListener("keydown", (e) => {
  if (state === "start" && e.key === "Enter") {
    state = "playing";
  } else if (state === "gameover" || state === "victory") {
    if (e.key === "r") resetGame();
    if (e.key === "e") state = "start";
  } else if (state === "playing") {
    if (e.key === "ArrowLeft") player.speed = -6;
    if (e.key === "ArrowRight") player.speed = 6;
    if (e.key === " ") {
      laser.x = player.x + player.w / 2;
      laser.y = player.y;
    }
  }
});

document.addEventListener("keyup", (e) => {
  if (state === "playing") {
    if (e.key === "ArrowLeft" || e.key === "ArrowRight") player.speed = 0;
  }
});

// Collision
function hit(a, b) {
  return (
    a.x < b.x + b.r &&
    a.x + a.w > b.x - b.r &&
    a.y < b.y + b.r &&
    a.y + a.h > b.y - b.r
  );
}

// Reset game
function resetGame() {
  player.x = 180;
  player.y = 380;
  laser.y = -20;
  player.speed = 0;
  invaderDX = 3;
  initInvaders();
  state = "playing";
}

// Draw spaceship
function drawPlayer() {
  let cx = player.x + player.w / 2;
  let cy = player.y + player.h / 2;

  // Body
  ctx.fillStyle = "white";
  ctx.beginPath();
  ctx.moveTo(cx, player.y);
  ctx.lineTo(player.x, player.y + player.h);
  ctx.lineTo(player.x + player.w, player.y + player.h);
  ctx.closePath();
  ctx.fill();

  // Cockpit
  ctx.fillStyle = "cyan";
  ctx.beginPath();
  ctx.arc(cx, cy, 5, 0, Math.PI * 2);
  ctx.fill();

  // Wings
  ctx.fillStyle = "gray";
  ctx.beginPath();
  ctx.moveTo(player.x, player.y + player.h);
  ctx.lineTo(player.x - 5, player.y + player.h + 10);
  ctx.lineTo(player.x + 10, player.y + player.h);
  ctx.closePath();
  ctx.fill();

  ctx.beginPath();
  ctx.moveTo(player.x + player.w, player.y + player.h);
  ctx.lineTo(player.x + player.w + 5, player.y + player.h + 10);
  ctx.lineTo(player.x + player.w - 10, player.y + player.h);
  ctx.closePath();
  ctx.fill();

  // Engine flame
  ctx.fillStyle = "orange";
  ctx.beginPath();
  ctx.moveTo(cx - 5, player.y + player.h);
  ctx.lineTo(cx + 5, player.y + player.h);
  ctx.lineTo(cx, player.y + player.h + 10);
  ctx.closePath();
  ctx.fill();
}

// Game loop
function update() {
  ctx.clearRect(0, 0, WIDTH, HEIGHT);

  if (state === "start") {
    ctx.fillStyle = "white";
    ctx.font = "30px Arial";
    ctx.fillText("SPACE INVADERS", 70, 150);
    ctx.font = "20px Arial";
    ctx.fillText("Press ENTER to Start", 100, 250);
  }

  if (state === "playing") {
    // Move player
    player.x += player.speed;
    player.x = Math.max(0, Math.min(WIDTH - player.w, player.x));

    // Move laser
    laser.y += laser.speed;

    // Move invaders
    let hitEdge = false;
    for (let inv of invaders) {
      inv.x += invaderDX;
      if (inv.x > WIDTH - 12 || inv.x < 12) hitEdge = true;
    }
    if (hitEdge) {
      invaderDX *= -1;
      for (let inv of invaders) inv.y += invaderDY;
    }

    // Laser collision
    invaders = invaders.filter(inv => {
      if (hit({ x: laser.x, y: laser.y, w: laser.w, h: laser.h }, inv)) return false;
      return true;
    });

    // Check win
    if (invaders.length === 0) state = "victory";

    // Player collision
    for (let inv of invaders) {
      if (hit(player, inv)) state = "gameover";
    }

    // Draw player
    drawPlayer();

    // Draw laser
    ctx.fillStyle = "lime";
    ctx.fillRect(laser.x, laser.y, laser.w, laser.h);

    // Draw invaders
    for (let inv of invaders) {
      ctx.fillStyle = inv.color;
      ctx.beginPath();
      ctx.arc(inv.x, inv.y, inv.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  if (state === "gameover") {
    ctx.fillStyle = "white";
    ctx.font = "40px Arial";
    ctx.fillText("GAME OVER", 100, 200);
    ctx.font = "20px Arial";
    ctx.fillText("Press R to Restart", 110, 250);
    ctx.fillText("Press E to Exit", 120, 280);
  }

  if (state === "victory") {
    ctx.fillStyle = "white";
    ctx.font = "40px Arial";
    ctx.fillText("YOU WIN!", 120, 200);
    ctx.font = "20px Arial";
    ctx.fillText("Press R to Restart", 110, 250);
    ctx.fillText("Press E to Exit", 120, 280);
  }

  requestAnimationFrame(update);
}

update();
